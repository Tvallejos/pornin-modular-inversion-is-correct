    requires { j - 1 >= 0 }
    requires { not_div = True }
    requires { not_div -> (xt a b f0 g0 (j-1)) >= 0 /\ (xt a b f1 g1 (j-1)) >= 0 } (* has not diverged *)
    requires { (xt a b f0 g0 (j-1)) = 0 \/ (regular' a b f0 g0 f1 g1 (j-1)) }
    requires { divides (factor_at (j-1)) (updated_x a b f0 g0) }
    requires { congr (xt a b f0 g0 (j - 1))
                      (xt (approx a (max (len a) (max (len b) (2 * k))) k) (approx b (max (len a) (max (len b) (2 * k))) k) f0 g0 (j - 1))

    ensures { xt a b f0 g0 ((j + 1) - 1) = 0 \/
         (len a + len b)
                   >= ((((len (xt a b f0 g0 ((j + 1) - 1))
                       + len (xt a b (2 * f1) (2 * g1) ((j + 1) - 1))) ) + (j + 1)) - 1) }
    ensures { xt a b f0 g0 ((j + 1) - 1) = 0 \/
         (regular' a b f0 g0 (2 * f1) (2 * g1) ((j + 1) - 1)) }
    = begin
    assert { not factor_at (j-1) = 0 };
    assert { divides ( 2 * factor_at (j - 1)) (updated_x a b f0 g0) };
    assert {   xt a b f0 g0 (j - 1) = 0 \/
  (len a + len b)
  >= ((len (xt a b f0 g0 (j - 1)) + len (xt a b f1 g1 (j - 1))) + (j - 1)) };
