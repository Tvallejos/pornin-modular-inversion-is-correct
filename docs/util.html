  lemma div2_no_div2 : forall a b. 
    divides 2 a -> not (divides 2 b) -> gcd a b = gcd (ED.div a 2) b   
    
  let lemma gcd_sub (a b : int)
    ensures { gcd a b = gcd (a - b) b }
  = begin 
    assert { gcd a b = gcd b a };
    assert { gcd b a = gcd b (a - 1 * b) };
  end 
  
  lemma gcd_odd_div2 : forall a b.
    odd a -> odd b -> gcd a b = gcd (ED.div (a - b) 2) b
  
  lemma gcd_mod21_div2 : forall a b.
    ED.mod a 2 = 1 -> ED.mod b 2 = 1 -> gcd a b = gcd (ED.div (a - b) 2) b
    


  lemma even_div_mult : forall n.
    ED.mod n 2 = 0 -> n = 2 * (ED.div n 2)
    
  lemma coprime_2_m : forall m.
    m >= 3 -> ED.mod m 2 = 1 -> coprime 2 m

  lemma coprime_sym : forall a b : int. coprime a b <-> coprime b a
