  let lemma mod_div (a b k m : int)
    requires { m <> 0 }
    requires { k <> 0 } 
    requires { congr a b m }
    requires { divides k a } 
    requires { divides k b } 
    requires { coprime k m }
    ensures { congr (ED.div a k) (ED.div b k) m }
   = begin 
     assert {exists q1. a = q1 * k};
     assert {exists q2. b = q2 * k};
     end
   meta "remove_unused:dependency" axiom mod_div, predicate congr

  let function div_mod (x m : int) : int 
    requires { m > 0 }
    requires { ED.mod m 2 = 1 }
    ensures { congr x (2 * result) m } 
  = 
  assert { congr x (2 * (x * (ED.div (m + 1) 2))) m 
    by 2 * (x * (ED.div (m + 1) 2)) = x * (m + 1) /\ congr x (m*x + x) m };
  x * (ED.div (m + 1) 2)
  
  let function mont_div_mod (x m : int) : int
    requires { m > 0 }
    requires { ED.mod m 2 = 1 }
    requires { 0 <= x < m }
    
    ensures { congr x (2 * result) m } 
    ensures { 0 <= result < m }
  = 
  if ED.mod x 2 = 0 
  then ED.div x 2 
  else (ED.div (x + m) 2)
