  let lemma div_sub (a b k : int)
   requires { k <> 0 } 
   requires { divides k a }
   requires { divides k b }
   ensures { ED.div (a - b) k = ED.div a k - ED.div b k }

    assert { exists q1 q2. a = q1 * k /\ b = q2 * k  /\ a - b = (q1 - q2) * k };
    end
