  type iter_vars =
    { a' : int;
      b' : int;
      k' : int;
      n' : int;
      j' : int;
      ba' : int;
      bb' : int;
      f0' : int;

  let function mk_iter (a b k n j ba bb f0 g0 f1 g1 :int) (ghost bf0 : int) (ghost bg0 : int) (ghost bf1 : int) (ghost bg1 : int) (bu bv : t) (ghost this_iter : bool) (divergence : bool) : iter_vars
  = {
    a' = a;
    b' = b;
    k' = k;
    n' = n;
    j' = j;
    ba' = ba;
    bb' = bb;
    f0' = f0;
    g0' = g0 ;
    f1' = f1 ;
    g1' = g1 ;
    bf0' = bf0;
    bg0' = bg0 ;
    bf1' = bf1;
    bg1' = bg1;
    bu' = bu ;
    bv' = bv;
    this_iter' = this_iter;
    divergence' = divergence;

    }

  let function a_t (vars : iter_vars) : int
      requires j_geq_1 { vars.j' - 1 >= 0 }
  = (xt vars.a' vars.b' vars.f0' vars.g0' (vars.j' - 1))

  let function b_t (vars : iter_vars) : int
      requires j_geq_1 { vars.j' - 1 >= 0 }
  = (xt vars.a' vars.b' vars.f1' vars.g1' (vars.j' - 1))

  let predicate is_halving' (ba : int) : bool =
    ED.mod ba 2 = 0

  let predicate is_halving (vars : iter_vars) : bool
  = is_halving' (vars.ba')

  let function halving_step (vars : iter_vars) : iter_vars
  =
  let ba = ED.div vars.ba' 2 in
  let bu = mont_div_mod vars.bu' in
  let bf0 = (div_mod vars.bf0' m) in
  let bg0 = (div_mod vars.bg0' m) in
  let f1 = 2*vars.f1' in
  let g1 = 2*vars.g1' in
  mk_iter vars.a' vars.b' vars.k' vars.n' vars.j' ba vars.bb' vars.f0' vars.g0' f1 g1 bf0 bg0 vars.bf1' vars.bg1' bu vars.bv' false vars.divergence'

  let function swap (vars : iter_vars) : iter_vars
  = mk_iter vars.a' vars.b' vars.k' vars.n' vars.j' vars.bb' vars.ba' vars.f1' vars.g1' vars.f0' vars.g0' vars.bf1' vars.bg1' vars.bf0' vars.bg0' vars.bv' vars.bu' vars.this_iter' vars.divergence'

  let predicate is_divergent' (a b j ba bb f0 g0 f1 g1 : int) : bool
  requires j_geq_1 { j - 1 >= 0 }
  =   (orb (andb (ba < bb) ((xt a b f0 g0 (j-1)) >= (xt a b f1 g1 (j-1)))) (andb (ba >= bb) ((xt a b f0 g0 (j-1)) < (xt a b f1 g1 (j-1)))))
  (*is_divergent (mk_iter a b k n j ba bb f0 g0 f1 g1 bf0 bg0 bf1 bg1 bu bv this_iter divergence)*)

  let predicate is_divergent (vars : iter_vars) : bool
    requires j_geq_1 { vars.j' - 1 >= 0 }
  =
  let ba = vars.ba' in let bb = vars.bb' in let a = vars.a' in let b = vars.b' in
  let f0 = vars.f0' in let g0 = vars.g0' in let f1 = vars.f1' in let g1 = vars.g1' in
  let j = vars.j' in
  is_divergent' a b j ba bb f0 g0 f1 g1

  lemma divergent_dec_w : forall vs : iter_vars. 0 <= j' vs - 1 -> is_divergent vs \/ ((not is_divergent vs))

  let predicate has_diverged (vars : iter_vars) : bool
  = vars.divergence'

  let predicate has_diverged' (a b k n j ba bb f0 g0 f1 g1 bf0 bg0 bf1 bg1 : int) (bu bv : t) (this_iter divergence : bool) : bool
  = has_diverged (mk_iter a b k n j ba bb f0 g0 f1 g1 bf0 bg0 bf1 bg1 bu bv this_iter divergence)

  let function set_this_iter (vars : iter_vars) : iter_vars
  = mk_iter vars.a' vars.b' vars.k' vars.n' vars.j' vars.ba' vars.bb' vars.f0' vars.g0' vars.f1' vars.g1' vars.bf0' vars.bg0' vars.bf1' vars.bg1' vars.bu' vars.bv' true vars.divergence'

  let function unset_this_iter (vars : iter_vars) : iter_vars
    ensures { result.this_iter' = false }
  = mk_iter vars.a' vars.b' vars.k' vars.n' vars.j' vars.ba' vars.bb' vars.f0' vars.g0' vars.f1' vars.g1' vars.bf0' vars.bg0' vars.bf1' vars.bg1' vars.bu' vars.bv' false vars.divergence'

  let function set_divergence (vars : iter_vars) : iter_vars
  = mk_iter vars.a' vars.b' vars.k' vars.n' vars.j' vars.ba' vars.bb' vars.f0' vars.g0' vars.f1' vars.g1' vars.bf0' vars.bg0' vars.bf1' vars.bg1' vars.bu' vars.bv' vars.this_iter' true

  let function inc_j (vars : iter_vars) : iter_vars
  ensures { result.a' = vars.a' }
  ensures { result.b' = vars.b' }
  ensures { result.k' = vars.k' }
  ensures { result.n' = vars.n' }
  ensures { result.j' = vars.j' + 1 }
  ensures { result.ba' = vars.ba' }
  ensures { result.bb' = vars.bb' }
  ensures { result.f0' = vars.f0' }
  ensures { result.g0' = vars.g0' }
  ensures { result.f1' = vars.f1' }
  ensures { result.g1' = vars.g1' }
  ensures { result.bf0' = vars.bf0' }
  ensures { result.bg0' = vars.bg0' }
  ensures { result.bf1' = vars.bf1' }
  ensures { result.bg1' = vars.bg1' }
  ensures { result.bu' = vars.bu' }
  ensures { result.bv' = vars.bv' }
  ensures { result.this_iter' = vars.this_iter' }
  ensures { result.divergence' = vars.divergence' }
  = mk_iter vars.a' vars.b' vars.k' vars.n' (vars.j' + 1) vars.ba' vars.bb' vars.f0' vars.g0' vars.f1' vars.g1' vars.bf0' vars.bg0' vars.bf1' vars.bg1' vars.bu' vars.bv' vars.this_iter' vars.divergence'

  let function update_divergence_status (vars : iter_vars) : iter_vars
    requires j_geq_1 { vars.j' - 1 >= 0 }
  =  if is_divergent vars then
      begin if has_diverged vars
            then vars (* divergence already set to true *)
            else set_this_iter (set_divergence vars)
      end
      else vars

  let lemma update_not_divergent (vars : iter_vars)
    requires j_geq_1 { vars.j' - 1 >= 0 }
    requires { not (is_divergent vars) }
    ensures { update_divergence_status vars = vars }
  = begin end

  let lemma update_divergent_has (vars : iter_vars)
    requires j_geq_1 { vars.j' - 1 >= 0 }
    requires { is_divergent vars }
    requires { has_diverged vars }
    ensures { update_divergence_status vars = vars }
  = begin end

  let lemma update_divergent_not_has (vars : iter_vars)
    requires j_geq_1 { vars.j' - 1 >= 0 }
    requires { is_divergent vars }
    requires { not (has_diverged vars) }
    ensures { update_divergence_status vars = set_this_iter (set_divergence vars)}
  = begin end

  let function odd_not_swap (vars : iter_vars) : iter_vars
  =
  let ba' = ED.div (vars.ba' - vars.bb') 2 in
  let bu' = mont_div_mod (sub vars.bu' vars.bv') in
  let f0' = vars.f0' - vars.f1' in
  let g0' = vars.g0' - vars.g1' in
  let bf0' = div_mod (vars.bf0' - vars.bf1') m in
  let bg0' = div_mod (vars.bg0' - vars.bg1') m in
  let f1' = 2*vars.f1' in
  let g1' = 2*vars.g1' in
  mk_iter vars.a' vars.b' vars.k' vars.n' vars.j' ba' vars.bb' f0' g0' f1' g1' bf0' bg0' vars.bf1' vars.bg1' bu' vars.bv' vars.this_iter' vars.divergence'

  let predicate is_ba_lt_bb (vars : iter_vars) : bool
  = vars.ba' < vars.bb'

  let predicate is_swap ( vars : iter_vars ) : bool
  = andb (vars.ba' < vars.bb') (not (is_halving vars))

  let predicate same_sign (x y : int)
  = orb (andb (x >= 0) (y >= 0)) (andb (x < 0) (y < 0))

  let predicate is_synchronous ( vars : iter_vars ) : bool
  requires j_geq_1 { vars.j' - 1 >= 0 }
  = let a = vars.a' in let b = vars.b' in let f0 = vars.f0' in let g0 = vars.g0' in
    let f1 = vars.f1' in let g1 = vars.g1' in let j = vars.j' in
  let a_t = xt a b f0 g0 (j - 1) in
  let b_t = xt a b f1 g1 (j - 1) in
  same_sign (b_t - a_t) (vars.bb' - vars.ba')

  let function odd_step (vars: iter_vars) : iter_vars
  requires j_geq_1 { vars.j' - 1 >= 0 }
  = let vars' = update_divergence_status vars in
  if is_ba_lt_bb vars'
    then odd_not_swap (swap vars')
    else odd_not_swap vars'

  let lemma update_swap (vars : iter_vars)
    requires j_geq_1 { vars.j' - 1 >= 0 }
    ensures { is_ba_lt_bb vars <-> is_ba_lt_bb (update_divergence_status vars) }
  = begin end

  let lemma odd_step_lt (vars: iter_vars)
  requires j_geq_1 { vars.j' - 1 >= 0 }
  requires { is_ba_lt_bb vars  }
  ensures { odd_step vars = odd_not_swap (swap (update_divergence_status vars)) }
  = begin end

  let lemma odd_step_not_lt (vars: iter_vars)
  requires j_geq_1 { vars.j' - 1 >= 0 }
  requires { not (is_ba_lt_bb vars) }
  ensures { odd_step vars = odd_not_swap (update_divergence_status vars) }
  = begin end

  let lemma this_iter_lt (vars : iter_vars)
   ensures { is_ba_lt_bb vars = is_ba_lt_bb (unset_this_iter vars) }
  = begin end

  let predicate decreasing (vars : iter_vars) : bool
   requires j_bounds { 1 <= vars.j' <= vars.k' - 1 }
  = let r = inc_j (inner_iter_fun' vars) in
  len (a_t r) + len (b_t r) <= len (a_t vars) + len (b_t vars) - 1

  let predicate regular (vars : iter_vars) : bool
      requires j_geq_1 { vars.j' - 1 >= 0 }
  = len (a_t vars) + len (b_t vars) <= len (vars.a') + len (vars.b') - (vars.j'-1)

  let predicate good (vars : iter_vars) : bool
      requires j_geq_1 { vars.j' - 1 >= 0 }
  = len (a_t vars) + len (b_t vars) <= len (vars.a') + len (vars.b') - (vars.k'-1)

   let predicate parity_approx (vars : iter_vars) : bool =
    requires { vars.k' > 1 }
    requires { vars.n' - vars.k' - 1 >= 0 }
    requires { (vars.j' - 1) >= 0 }
    parity_approx' vars.a' vars.b' vars.f0' vars.g0' vars.n' vars.j' vars.k'

  let lemma parity_approx_even (vars : iter_vars)
    requires { vars.k' > 1 }

    requires { 1 <= vars.j' <= vars.k' - 1 }
    ensures { parity_approx vars }
